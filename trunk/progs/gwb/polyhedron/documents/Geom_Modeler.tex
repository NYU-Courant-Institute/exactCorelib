\begin{abstract}

\noindent The problem of computing geometry modeling in 3D is fundamental to computer graphics, path finding, motion planning and numerous branches of computer science. Given a polyhedron P defined by CSG (Constructive Solid Geometry) model, we raise a new idea to present the boundary features (vertices, edges and faces) of P using subdivision method; it has the following steps: 1) define a polyhedron P in CSG model, 2) predicate and subdivide a given box $B_0$ recursively until all sub-boxes are elementary 3) generate the adequate Subdivision Tree, 4) represent the boundary features based on the Subdivision Tree. Since the great generality and scalability of subdivision method, this idea can also be expanded to support more complex shape with higher order surfaces in 3D.

\end{abstract}

\subsection {Subdivision on box $B$}
Suppose $B=([x_1,x_2],[y_1,y_2],[z_1,z_2])$, let $\bar{x} = \frac {x_1+x_2}{2}$, $\bar{y} = \frac {y_1+y_2}{2}$, $\bar{z} = \frac {z_1+z_2}{2}$. The subdivision on $B$ is to subdivide to box $B$ into eight smaller boxes:

\begin{itemize}
	\item $B_0=([x_1,\bar{x}],[y_1,\bar{y}],[z_1,\bar{z}])$
	\item $B_1=([x_1,\bar{x}],[y_1,\bar{y}],[\bar{z},z_2])$
	\item $B_2=([x_1,\bar{x}],[\bar{y},y_2],[z_1,\bar{z}])$
	\item $B_3=([x_1,\bar{x}],[\bar{y},y_2],[\bar{z},z_2])$
	\item $B_4=([\bar{x},x_2],[y_1,\bar{y}],[z_1,\bar{z}])$
	\item $B_5=([\bar{x},x_2],[y_1,\bar{y}],[\bar{z},z_2])$
	\item $B_6=([\bar{x},x_2],[\bar{y},y_2],[z_1,\bar{z}])$
	\item $B_7=([\bar{x},x_2],[\bar{y},y_2],[\bar{z},z_2])$
\end{itemize}	

We name $B_i\ (i=0,1,2,\ldots,7)$ a subbox of $B$.

\subsection{Elementary Box}
%------------elementary boxes---------------------
	Then we define the \textbf{elementary box} $B_e$ for a polyhedron $P$: 
		\begin{enumerate} 
           		\item \textbf{Empty Box}:
           			 a box $B_e$ is an empty box $\mathrm{iff}$ $B_e\cap P=\emptyset$.
           		\item \textbf{Full Box}: 
            			a box $B_e$ is a full box of Polyhedron $P$ $\mathrm{iff}$ $B_e\cap P = B_e$
            		\item \textbf{Vertex Box}: 
           			 a box $B_e$ is a vertex box $\mathrm{iff}$  there exists only one vertex $v \in V \cap B_e$, s.t.\ $\forall e \in E$, if $e \cap B_e \not= \emptyset$, $v$ bounds $e$, and $\forall f \in F$, if $f \cap B_e \not= \emptyset$, $v$ bounds $F$. 
            		\item \textbf{Edge Box}: 
            			a box $B_e$ is an edge box $\mathrm{iff}$ $V \cap B_e = \emptyset$ and there exists only one edge $ e\in E$, $e\cap B_e \not= \emptyset, \forall f \in F$, if $f \cap B_e \not= \emptyset$, e bounds f. 
            		\item \textbf{Face Box}:
            			a box $B_e$ is a face box $\mathrm{iff}$ $V \cap B_e = \emptyset$, $E \cap B_e = \emptyset$ and there exists only one face $f \in F$, and $f \cap B_e \not= \emptyset$. 
                  \end{enumerate}
\subsection{Defination of S-Tree and AS-Tree}
\begin{enumerate}

%subdivision tree node
	\item Subdivision Tree Node (ST-Node) for box $B$ and polyhedron $P$, represented as $n(B,P)$ : a Subdivision Tree Node is a node contains the following information:
		\begin{itemize}
			\item pointer to the parent of $n$: $pp$
			\item a list of pointers to the eight children of node $n$: $p_i ( i=0,1,2, \ldots, 7)$ 
			\item a box $B$
			\item  $\Phi^*\ (B, P)$
		\end{itemize}
	%\indent We represent the Subdivision Tree box $B$ and polyhedron $P$ as $n(B,P)$.\\
	\indent If $B$ is an Elementary Box for $P$ in node $n(B,P)$, then $n(B,P)$ is called an Elementary Subdivistion Tree Node (EST-Node). 
	\indent Subdivision on a Subdivision Tree Node $n(B,P)$ is defined as :
		\begin{enumerate}
			\item Subdivide box $B$ into 8 subboxes $(B_0, B_1,\ldots ,B_7)$
			\item Compute $\Phi^*(B_i, P)(i=0,1,2,\ldots,7)$
			\item Construct eight ST-Nodes $n(B_i,P)$ and set each $pp$ of $n(B_i,P)$ point to $B$
			\item Set $p_i$ of $n(B,P)$ point to $n(B_i,P)\ (i=0,1,2,\ldots,7)$

		\end{enumerate}


%subdivision tree
	\item Subdivision Tree (S-Tree) for box $B_s$ and polyhedron $P$: A Subdivision Tree (S-Tree ) is  an octree rooted at $n(B_s,P)$ and in which each node is a Subdivision Tree Node for a box $B$ and polyhedron $P$ . For each inner node $n(B,P)$,  the eight children of $n(B,P)$ are $n(B_i,P)\ (i=0,1,2,\ldots, 7)$. 
	\item Adequate Subdivision Tree (AS-Tree) for box $B_s$ and polyhedron $P$ : An Adequate Subdivision Tree(AS-Tree) for box $B_s$ and polyhedron $P$ is a S-Tree whose leaves are EST- Nodes.\\
\indent We represent an AS-Tree for box $B_s$ and polyhedron $P$ as $AST(B_s,P)$
	\item Simplest Adequate Subdivision Tree (SAS-Tree) for a box $B_s$ and polyhedron $P$ : 
		\begin{enumerate}
			\item Construct a ST-node $n(B_s,P)$ and let it be the root of S-Tree.
			\item Subdivide $n(B_s,P)$ if it is not an EST-node.
			\item Keep subdividing every leaf of the S-Tree until it is an AS-Tree, which means that every leaf of the S-Tree is an EST-Node.
			\item Now the AS-Tree we have is an SAS-Tree.
Finally we get an AS-Tree. This tree is an SAS-Tree. \\
We represent an SAS-Tree constructed by box $B_s$ and polyhedron $P$ as $SAST(B_s,P)$.

		\end{enumerate}

\end{enumerate}
